    \section{实验目的}
    
    本次实验围绕操作系统的两个核心主题——进程管理（侧重于优先级）和进程同步（侧重于信号量）展开，旨在达成以下目标：
    
    \subsection{深入理解进程管理与调度}
    学习如何在操作系统内核中实现进程优先级的查看和修改机制。理解优先级如何作为进程调度的重要依据，为从默认的轮转调度（Round Robin）转向优先级调度奠定基础。
    
    \subsection{掌握内核数据结构的扩展与系统调用}
    学习如何修改关键的内核数据结构，特别是进程控制块（\texttt{struct proc}），以包含进程优先级信息；学习如何定义新的内核数据结构来实现信号量（\texttt{struct semaphore}）。同时，通过实现 \texttt{getptable()}、\texttt{setpriority()}、\texttt{sem\_init()}等一系列系统调用，熟练掌握从用户态发起请求到内核态响应处理的完整流程。
    
    \subsection{深入理解进程同步与互斥}
    分析多进程并发访问共享资源（如文件）时可能出现的竞态条件（Race Condition）和数据不一致问题。理解临界区（Critical Section）的概念以及保护临界区的必要性。
    
    \subsection{掌握信号量机制的原理与实现}
    学习 Dijkstra 提出的信号量机制（Semaphore），理解 P 操作（\texttt{wait}）和 V 操作（\texttt{signal}）的原理。掌握如何在内核中利用 \texttt{sleep()} 和 \texttt{wakeup()} 原语来实现信号量的阻塞和唤醒功能。

    \section{实验要求}
    
    \subsection{任务一：进程优先级管理}
    \begin{enumerate}
        \item \textbf{实现 \texttt{ps} 命令 (ps.c)}：显示系统进程列表，包括 PID, PPID, PRI (优先级), MEM, STATE, CMD。
        \item \textbf{实现系统调用 \texttt{getptable()}}：用于从内核获取进程表信息。
        \item \textbf{实现 \texttt{nice} 命令 (nice.c)}：用户态程序，用于修改进程优先级。命令格式：\texttt{nice pid priority}（优先级范围 [1, 20]，数值越小优先级越高）。
        \item \textbf{实现系统调用 \texttt{setpriority(pid, priority)}}：供 \texttt{nice} 命令调用。
    \end{enumerate}
    
    \subsection{任务二：信号量实现}
    在内核中实现以下系统调用：
    \begin{enumerate}
        \item \texttt{int sem\_init(int sem, int value)}：初始化信号量 \texttt{sem} 的值为 \texttt{value}。
        \item \texttt{int sem\_destroy(int sem)}：销毁信号量 \texttt{sem}。
        \item \texttt{int sem\_wait(int sem, int count)}：P 操作，申请 \texttt{count} 个资源，如果不足则阻塞。
        \item \texttt{int sem\_signal(int sem, int count)}：V 操作，释放 \texttt{count} 个资源，并唤醒等待的进程。
    \end{enumerate}
    
    \subsection{任务三：使用信号量解决文件读写互斥}
    \begin{enumerate}
        \item 创建 \texttt{sem\_test.c}。
        \item 启动 10 个并发进程。
        \item 每个进程重复 50 次以下操作：从文件中读取数值，加 1，写回文件。
        \item 使用任务二实现的信号量保证操作的互斥性，验证最终文件中的数值为 500。
    \end{enumerate}

    \subsubsection{步骤一：定义信号量数据结构与初始化}
    在内核中定义信号量结构体，并创建全局信号量数组。
    
    \begin{codeblock}{信号量数据结构定义}
        #include "spinlock.h"
        #define MAX_SEMAPHORES 16 // 系统最大支持的信号量数量
        
        // (定义在 proc.h 或相关头文件中)
        struct semaphore {
            int value;            // 信号量的计数值
            struct spinlock lock; // 保护信号量的自旋锁
            int active;           // 标记该信号量是否正在使用
        };
        
        // (在 proc.c 中定义全局数组)
        struct semaphore semaphores[MAX_SEMAPHORES];
    \end{codeblock}
    
    在系统启动时（例如在 \texttt{main.c} 中调用 \texttt{seminit()}），需要初始化这些信号量的锁。

        int
        sys_sem_init(void)
        {
            int sem, value;
            // ... (参数获取与校验) ...
            
            // 初始化操作
            acquire(&semaphores[sem].lock);
            semaphores[sem].value = value;
            semaphores[sem].active = 1;
            release(&semaphores[sem].lock);
            
            return 0;
        }
    \end{codeblock}

    \subsubsection{步骤三：实现 sem\_wait() (P操作)}
    \texttt{sem\_wait} 是核心操作。如果资源不足，进程必须阻塞。关键点在于使用 \texttt{while} 循环检查条件，并调用 \texttt{sleep(chan, lock)} 进行阻塞。
    
    \begin{codeblock}{sysproc.c (sys\_sem\_wait)}
        int
        sys_sem_wait(void)
        {
            int sem, count;
            // ... (参数获取与校验) ...
            
            acquire(&semaphores[sem].lock);
            
            // 当资源不足时，阻塞等待 (使用 while 而非 if 是关键)
            while(semaphores[sem].value < count){
                // 使用信号量的地址作为等待通道 (chan)
                // sleep 会原子性地释放 lock 并使当前进程进入 SLEEPING 状态
                sleep(&semaphores[sem], &semaphores[sem].lock);
                // 被唤醒后，sleep 会重新获取 lock
            }
            
            // 资源足够，分配资源
            semaphores[sem].value -= count;
            
            release(&semaphores[sem].lock);
            
            return 0;
        }
    \end{codeblock}

        // setpriority：修改指定进程的优先级
        // 参数：pid - 目标进程ID，priority - 新优先级值(1-20)
        // 返回值：0表示成功，-1表示失败
        int
        setpriority(int pid, int priority)
        {
            struct proc *p;          // 用于遍历进程表的指针
            int found = 0;           // 标志：是否找到目标进程
            
            // 步骤1：参数校验 - 优先级必须在[1, 20]范围内
            if(priority < 1 || priority > 20)
            return -1;
            
            // 步骤2：获取进程表锁，确保互斥访问
            acquire(&ptable.lock);
            
            // 步骤3：遍历进程表，查找目标进程
            for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
                if(p->pid == pid){
                    p->priority = priority;  // 修改优先级
                    found = 1;
                    break;
                }
            }
            
            // 步骤4：释放进程表锁
            release(&ptable.lock);
            
            // 步骤5：返回结果
            return found ? 0 : -1;
        }
    \end{codeblock}
    
    \subsubsection{B.2 proc.c - getptable函数}
    
    \begin{codeblock}{getptable实现}
        // getptable：将内核进程表信息复制到用户空间
        // 参数：ubuf - 用户空间缓冲区地址，size - 缓冲区大小
        // 返回值：0表示成功，-1表示失败
        int 
        getptable(void *ubuf, int size)
        {
            struct proc *p;              // 进程表遍历指针
            struct proc_info pi;         // 临时变量，用于构造进程信息
            char *uptr = (char*)ubuf;    // 用户缓冲区指针
            int i = 0;                   // 进程计数器
            
            // 步骤1：缓冲区大小检查
            if(size < NPROC * sizeof(struct proc_info))
            return -1;
            
            // 步骤2：获取进程表锁
            acquire(&ptable.lock);
            
            // 步骤3：遍历进程表
            for(p = ptable.proc; p < &ptable.proc[NPROC]; p++){
                // 清零临时结构体
                memset(&pi, 0, sizeof(pi));
                
                // 如果槽位正在使用，填充信息
                if(p->state != UNUSED){
                    pi.pid = p->pid;
                    pi.ppid = (p->parent) ? p->parent->pid : -1; 
                    pi.priority = p->priority;
                    pi.mem_size = p->sz;
                    pi.state = p->state;
                    safestrcpy(pi.name, p->name, sizeof(pi.name));
                }
                
                // 步骤4：复制到用户空间
                if(copyout(myproc()->pgdir, 
                (uint)(uptr + i * sizeof(struct proc_info)), 
                (char*)&pi, sizeof(pi)) < 0){
                    release(&ptable.lock);
                    return -1;
                }
                i++;
            }
            
            // 步骤5：释放锁并返回
            release(&ptable.lock);
            return 0;
        }
    \end{codeblock}

    \begin{codeblock}{sem\_init实现}
        // sem_init：初始化一个信号量
        // 参数：sem - 信号量ID(0-31)，value - 信号量初始值
        // 返回值：0表示成功，-1表示失败
        int
        sem_init(int sem, int value)
        {
            // 步骤1：参数校验
            if(sem < 0 || sem >= 32)
            return -1;
            
            // 步骤2：获取该信号量的锁
            acquire(&sema[sem].lock);
            
            // 步骤3：检查是否已被初始化
            if(sema[sem].active == 1) {
                release(&sema[sem].lock);
                return -1;  // 已在使用，不允许重复初始化
            }
            
            // 步骤4：初始化信号量
            sema[sem].value = value;  // 设置初始值
            sema[sem].active = 1;     // 标记为活跃状态
            
            // 步骤5：释放锁
            release(&sema[sem].lock);
            
            return 0;
        }
    \end{codeblock}

    \begin{codeblock}{sem\_destroy实现}
        // sem_destroy：销毁一个信号量
        // 参数：sem - 信号量ID(0-31)
        // 返回值：0表示成功，-1表示失败
        int
        sem_destroy(int sem)
        {
            // 步骤1：参数校验
            if(sem < 0 || sem >= 32)
            return -1;
            
            // 步骤2：获取锁
            acquire(&sema[sem].lock);
            
            // 步骤3：检查是否处于活跃状态

            sema[sem].active = 0;  // 标记为未使用
            sema[sem].value = 0;   // 清零计数值
            
            // 步骤5：释放锁
            release(&sema[sem].lock);
            
            return 0;
        }

        {
            // 步骤1：参数校验
            if(sem < 0 || sem >= 32)
            return -1;
            
            // 步骤2：获取信号量锁
            acquire(&sema[sem].lock);
            
            // 步骤3：检查是否处于活跃状态
            if(sema[sem].active == 0){
                release(&sema[sem].lock);
                return -1;
            }
            
            // 步骤4：循环检查资源是否足够（核心逻辑）
            // 使用while而非if，防止虚假唤醒
            while(sema[sem].value < count){
                // 资源不足，进程进入睡眠
                // sleep的参数1：睡眠通道（使用信号量地址）
                // sleep的参数2：当前持有的锁
                // sleep会原子性地释放锁并进入SLEEPING状态
                sleep(&sema[sem], &sema[sem].lock);
                
                // 被唤醒后，回到while开头重新检查条件
                
                // 防御性检查：信号量是否在睡眠期间被销毁
                if(sema[sem].active == 0){
                    release(&sema[sem].lock);
                    return -1;
                }
            }
            
            // 步骤5：资源充足，消耗资源
            sema[sem].value -= count;
            
            // 步骤6：记账，用于进程退出时的自动资源回收
            myproc()->sem_held[sem] += count;
            
            // 步骤7：释放锁
            release(&sema[sem].lock);
            
            return 0;
        }
    \end{codeblock}
    
    \subsubsection{D.4 proc.c - sem\_signal函数(V操作)}
    
    \begin{codeblock}{sem\_signal实现}
        // sem_signal：V操作，释放资源并唤醒等待者
        // 参数：sem - 信号量ID，count - 需要释放的资源数量
        // 返回值：0表示成功，-1表示失败
        int 
        sem_signal(int sem, int count)
        {
            // 步骤1：参数校验
            if(sem < 0 || sem >= 32)
            return -1;
            
            // 步骤2：获取信号量锁
            acquire(&sema[sem].lock);
            
            // 步骤3：检查活跃状态
            if(sema[sem].active == 0){
                release(&sema[sem].lock);
                return -1;
            }
            
            // 步骤4：释放资源，增加信号量的值
            sema[sem].value += count;
            
            // 步骤5：销账，减少当前进程的资源持有计数
            if(myproc()->sem_held[sem] >= count) {
                myproc()->sem_held[sem] -= count;
            } else {
                // 防御性处理：持有计数不足，清零
                myproc()->sem_held[sem] = 0;
            }
            
            // 步骤6：唤醒所有等待该信号量的进程
            // wakeup将所有在该通道上睡眠的进程状态改为RUNNABLE
            wakeup(&sema[sem]);
            
            // 步骤7：释放锁
            release(&sema[sem].lock);
            
            return 0;
        }
    \end{codeblock}

    \subsubsection{E.1 sem\_test.c - 完整实现}
    
    \begin{codeblock}{sem\_test.c}
        #include "types.h"
        #include "stat.h"
        #include "user.h"
        #include "fcntl.h"
        
        // 测试参数定义
        #define NUM_CHILDREN 10              // 并发子进程数量
        #define TARGET_COUNT_PER_CHILD 50    // 每个子进程的循环次数
        #define COUNTER_FILE "counter"       // 共享计数器文件名
        #define SEM_ID 0                     // 使用0号信号量
        
        // test_counter：信号量互斥测试函数
        void
        test_counter()
        {
            int i, j;
            int fd;
            int val;
            
            // ===== 第一部分：初始化 =====
            
            // 步骤1：创建共享计数器文件并写入初始值0
            fd = open(COUNTER_FILE, O_CREATE | O_RDWR);
            if(fd < 0){
                printf(1, "Error: create counter file failed\n");
                exit();
            }
            val = 0;
            write(fd, &val, sizeof(int));  // 写入4字节整数
            close(fd);
            
            // 步骤2：初始化互斥信号量（初始值为1，作为互斥锁）
            if(sem_init(SEM_ID, 1) < 0){
                printf(1, "Error: sem_init failed\n");
                exit();
            }
            
            printf(1, "Starting %d children, each incrementing %d times...\n", 
            NUM_CHILDREN, TARGET_COUNT_PER_CHILD);
            
            // ===== 第二部分：创建并发子进程 =====
            
            for(i = 0; i < NUM_CHILDREN; i++){
                int pid = fork();
                if(pid < 0){
                    printf(1, "fork failed\n");
                    exit();
                }
                
                if(pid == 0){
                    // --- 子进程代码 ---
                    for(j = 0; j < TARGET_COUNT_PER_CHILD; j++){
                        
                        // P操作：申请进入临界区
                        if(sem_wait(SEM_ID, 1) < 0){
                            printf(1, "Child: sem_wait failed\n");
                            exit();
                        }
                        
                        // ========== 临界区开始 ==========
                        
                        // 读取当前计数值
                        fd = open(COUNTER_FILE, O_RDONLY);
                        if(fd < 0){
                            printf(1, "Child: open read failed\n");
                            exit();
                        }
                        if(read(fd, &val, sizeof(int)) != sizeof(int)){
                            printf(1, "Child: read failed\n");
                            exit();
                        }
                        close(fd);
                        
                        // 更新计数值（加1）
                        val++;
                        
                        // 写回新值
                        fd = open(COUNTER_FILE, O_WRONLY); 
                        if(fd < 0){
                            printf(1, "Child: open write failed\n");
                            exit();
                        }
                        if(write(fd, &val, sizeof(int)) != sizeof(int)){
                            printf(1, "Child: write failed\n");
                            exit();
                        }
                        close(fd);
                        
                        // ========== 临界区结束 ==========
                        
                        // V操作：离开临界区
                        if(sem_signal(SEM_ID, 1) < 0){
                            printf(1, "Child: sem_signal failed\n");
                            exit();
                        }
                    }
                    
                    exit();  // 子进程完成后退出
                }
            }
            
            // ===== 第三部分：等待子进程并验证结果 =====
            
            // 父进程等待所有子进程退出
            for(i = 0; i < NUM_CHILDREN; i++){
                wait();
            }
            
            // 读取最终计数值
            fd = open(COUNTER_FILE, O_RDONLY);
            read(fd, &val, sizeof(int));
            close(fd);
            
            printf(1, "Final counter value: %d\n", val);
            
            // 验证结果（期望值：10 × 50 = 500）
            if(val == NUM_CHILDREN * TARGET_COUNT_PER_CHILD){
                printf(1, "TEST PASSED!\n");
            } else {
                printf(1, "TEST FAILED! Expected %d\n", 
                NUM_CHILDREN * TARGET_COUNT_PER_CHILD);
            }
            
            // ===== 第四部分：清理资源 =====
            
            sem_destroy(SEM_ID);    // 销毁信号量
            unlink(COUNTER_FILE);   // 删除测试文件
            
            return;
        }
        
        int
        main(int argc, char *argv[])
        {
            test_counter();
            return 0;
        }
    \end{codeblock}
